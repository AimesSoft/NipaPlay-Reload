cmake_minimum_required(VERSION 3.14)
project(nipa_media_probe LANGUAGES CXX)
add_executable(nipa_media_probe main.cpp)
target_compile_features(nipa_media_probe PRIVATE cxx_std_17)
if(WIN32)
  target_compile_options(nipa_media_probe PRIVATE /utf-8)
else()
  target_link_libraries(nipa_media_probe PRIVATE ${CMAKE_DL_LIBS})
endif()
